Question 2: Write a program to perform Breadth First Search (BFS) traversal on the graph {A, B, C, D, E, F}.
(Exact code from your BFS screenshot)
from collections import deque
graph = {
    'A': ['B', 'C'],
    'B': ['D'],
    'C': ['E'],
    'D': ['F'],
    'E': [],
    'F': []
}
def bfs(graph, start):
    visited = set()
    queue = deque([start])
    visited.add(start)
    while queue:
        vertex = queue.popleft()
        print(vertex, end=" ")
        for neighbour in graph[vertex]:
            if neighbour not in visited:
                visited.add(neighbour)
                queue.append(neighbour)

print("BFS traversal:")
bfs(graph, 'A')

def merge_sort(l1):
    if len(l1) > 1:
        mid = len(l1) // 2
        left = l1[:mid]
        right = l1[mid:]

        merge_sort(left)
        merge_sort(right)

        i = j = k = 0
        while i < len(left) and j < len(right):
            if left[i] < right[j]:
                l1[k] = left[i]
                i += 1
            else:
                l1[k] = right[j]
                j += 1
            k += 1

        while i < len(left):
            l1[k] = left[i]
            i += 1
            k += 1

        while j < len(right):
            l1[k] = right[j]
            j += 1
            k += 1

Question 1: Write a Python program to implement Quick Sort (last element as pivot).
def partition(l1, low, high):
    pivot = l1[high]
    i = low - 1
    for j in range(low, high):
        if l1[j] <= pivot:
            i = i + 1
            l1[i], l1[j] = l1[j], l1[i]
    l1[i + 1], l1[high] = l1[high], l1[i + 1]
    return i + 1

def quick_sort(l1, low, high):
    if low < high:
        pi = partition(l1, low, high)
        quick_sort(l1, low, pi - 1)
        quick_sort(l1, pi + 1, high)

l1 = []
n = int(input("enter size:"))
for i in range(n):
    l1.append(int(input("enter element:")))

quick_sort(l1, 0, n - 1)
print("Sorted list in ascending order:", l1)

Question 2: Write a program to perform Breadth First Search (BFS) traversal on the graph {A, B, C, D, E, F}.
(Same as Slip 2, Question 2)
from collections import deque
graph = {
    'A': ['B', 'C'],
    'B': ['D'],
    'C': ['E'],
    'D': ['F'],
    'E': [],
    'F': []
}
def bfs(graph, start):
    visited = set()
    queue = deque([start])
    visited.add(start)
    while queue:
        vertex = queue.popleft()
        print(vertex, end=" ")
        for neighbour in graph[vertex]:
            if neighbour not in visited:
                visited.add(neighbour)
                queue.append(neighbour)

print("BFS traversal:")
bfs(graph, 'A')
